    commands:

      - |

        import pandas as pd

        import sys



        print("=== Lecture des sources BottleNeck ===")



        erp = pd.read_excel("/data/Fichier_erp.xlsx")

        liaison = pd.read_excel("/data/fichier_liaison.xlsx")

        web = pd.read_excel("/data/Fichier_web.xlsx")



        # --------------------------------------------------------

        # Controle du schema pour detecter une nouvelle source

        # incompatible avant de lancer les traitements

        # --------------------------------------------------------



        required_erp = {

            "product_id",

            "onsale_web",

            "price",

            "stock_quantity",

            "stock_status"

        }



        required_liaison = {

            "product_id",

            "id_web"

        }



        required_web = {

            "sku",

            "total_sales",

            "post_title",

            "post_name",

            "post_type"

        }



        missing = []



        if not required_erp.issubset(erp.columns):

            missing.append(

                "ERP colonnes manquantes : "

                + str(sorted(required_erp - set(erp.columns)))

            )



        if not required_liaison.issubset(liaison.columns):

            missing.append(

                "Liaison colonnes manquantes : "

                + str(sorted(required_liaison - set(liaison.columns)))

            )



        if not required_web.issubset(web.columns):

            missing.append(

                "WEB colonnes manquantes : "

                + str(sorted(required_web - set(web.columns)))

            )



        if missing:

            print("ECHEC VALIDATION DES SOURCES")



            for message in missing:

                print("-", message)



            sys.exit(1)



        # --------------------------------------------------------

        # Normalisation des cles

        # --------------------------------------------------------



        def normaliser_cle(serie, prefixe):

            s = serie.astype("string")

            s = s.str.strip()

            s = s.str.lower()

            s = s.str.replace(

                r"\.0$",

                "",

                regex=True

            )



            s = s.replace({

                "": pd.NA,

                "nan": pd.NA,

                "<na>": pd.NA,

                "none": pd.NA

            })



            return s.apply(

                lambda x:

                prefixe + x

                if pd.notna(x)

                else pd.NA

            )



        # ERP <-> Liaison

        erp["product_key"] = normaliser_cle(

            erp["product_id"],

            "p_"

        )



        liaison["product_key"] = normaliser_cle(

            liaison["product_id"],

            "p_"

        )



        # Liaison <-> Web

        liaison["web_key"] = normaliser_cle(

            liaison["id_web"],

            "w_"

        )



        web["web_key"] = normaliser_cle(

            web["sku"],

            "w_"

        )



        print(

            f"ERP brut : {len(erp)} lignes"

        )



        print(

            f"Liaison brute : {len(liaison)} lignes"

        )



        print(

            f"WEB brut : {len(web)} lignes"

        )



        erp.to_csv(

            "erp_raw.csv",

            index=False

        )



        liaison.to_csv(

            "liaison_raw.csv",

            index=False

        )



        web.to_csv(

            "web_raw.csv",

            index=False

        )



        print(

            "Sources validees."

        )



        print(

            "Conversion Excel -> CSV terminee."

        )

        PY



    outputFiles:

      - "erp_raw.csv"

      - "liaison_raw.csv"

      - "web_raw.csv"

  # 20. CALCUL DU Z-SCORE

  #

  # z = (prix - moyenne) / ecart-type

  # premium / millesime si z > 2

  # ============================================================



  - id: compute_zscore

    type: io.kestra.plugin.scripts.shell.Commands



    taskRunner:

      type: io.kestra.plugin.core.runner.Process



    inputFiles:

      merged.csv: "{{ outputs.export_merged_csv.uri }}"



    commands:

      - |

        /usr/bin/python3 - <<'PY'

        import pandas as pd



        df = pd.read_csv(

            "{{ workingDir }}/merged.csv"

        )



        df["price"] = pd.to_numeric(

            df["price"],

            errors="coerce"

        )



        moyenne = df["price"].mean()



        ecart_type = df["price"].std()



        df["zscore"] = (

            df["price"] - moyenne

        ) / ecart_type



        df["categorie"] = (

            df["zscore"]

            .apply(

                lambda z:

                "premium"

                if z > 2

                else "ordinaire"

            )

        )



        df.to_csv(

            "zscore.csv",

            index=False

        )



        nb_premium = int(

            (

                df["zscore"] > 2

            ).sum()

        )



        nb_ordinaires = int(

            (

                df["zscore"] <= 2

            ).sum()

        )



        print(

            f"Prix moyen : "

            f"{moyenne:.2f} EUR"

        )



        print(

            f"Ecart-type : "

            f"{ecart_type:.2f} EUR"

        )



        print(

            "Vins premium / millesimes "

            f"(z-score > 2) : "

            f"{nb_premium}"

        )



        print(

            f"Vins ordinaires : "

            f"{nb_ordinaires}"

        )

        PY



    outputFiles:

      - "zscore.csv"
