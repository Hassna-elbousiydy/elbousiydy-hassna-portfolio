// -------------------------------------------------------
// B1. Taux de réservation moyen par mois par type de logement
// -------------------------------------------------------
// Explication : Le champ reviews_per_month est le proxy standard utilisé par
// InsideAirbnb pour estimer le taux de réservation (= activité du logement).
// On calcule la moyenne de reviews_per_month par room_type,
// en excluant les documents sans valeur (null).

db.listings.aggregate([
  {
    $match: {
      reviews_per_month: { $exists: true, $ne: null, $type: "number" }
    }
  },
  {
    $group: {
      _id: "$room_type",
      taux_moyen_mois: { $avg: "$reviews_per_month" },
      nb_logements: { $sum: 1 }
    }
  },
  {
    $sort: { taux_moyen_mois: -1 }
  },
  {
    $project: {
      _id: 0,
      type_logement: "$_id",
      taux_moyen_mois: { $round: ["$taux_moyen_mois", 3] },
      nb_logements: 1
    }
  }
]);

/* RÉSULTATS ATTENDUS :

// -------------------------------------------------------
// B2. Médiane des nombre d'avis pour tous les logements
// -------------------------------------------------------
// Explication : MongoDB n'a pas de $median natif avant la version 7.
// Méthode : tri → positionnement à 50% → extraction de la valeur.
// Alternative Python/Polars recommandée pour les grandes collections.

db.listings.aggregate([
  {
    $sort: { number_of_reviews: 1 }
  },
  {
    $group: {
      _id: null,
      all_reviews: { $push: "$number_of_reviews" },
      count: { $sum: 1 }
    }
  },
  {
    $project: {
      median: {
        $arrayElemAt: [
          "$all_reviews",
          { $floor: { $divide: ["$count", 2] } }
        ]
      }
    }
  }
]);

/* RÉSULTAT ATTENDU : 3 avis (médiane)

   ⚠️ Interprétation : La médiane à 3 signifie que 50% des logements
   ont reçu 3 avis ou moins. Cela reflète une forte asymétrie :
   de nombreux logements peu actifs, quelques uns très actifs (jusqu'à 3 067 avis).
   La moyenne (non représentative) serait bien plus élevée à cause des outliers.
*/

// -------------------------------------------------------
// B3. Médiane des avis par catégorie d'hôte (super hôte vs non super hôte)
// -------------------------------------------------------
// Méthode : on trie par catégorie + avis, puis on extrait la médiane par groupe.
// Note : MongoDB <= 6 requiert un contournement avec $push + $arrayElemAt

db.listings.aggregate([
  {
    $group: {
      _id: "$host_is_superhost",
      all_reviews: { $push: "$number_of_reviews" },
      count: { $sum: 1 }
    }
  },
  {
    $project: {
      categorie: {
        $cond: { if: { $eq: ["$_id", "t"] }, then: "Super hôte", else: "Hôte standard" }
      },
      count: 1,
      median_reviews: {
        $arrayElemAt: [
          { $sortArray: { input: "$all_reviews", sortBy: 1 } },
          { $floor: { $divide: ["$count", 2] } }
        ]
      }
    }
  },
  {
    $sort: { median_reviews: -1 }
  }
]);

/* RÉSULTATS ATTENDUS :
   Super hôte     : médiane = 24 avis
   Hôte standard  : médiane =  2 avis

   ⚠️ Interprétation clé : Les super hôtes ont 12x plus d'avis
   en médiane → preuve d'une bien plus forte activité/qualité.
   Le statut "super hôte" est corrélé à un nombre élevé de réservations.
   Cela peut indiquer que ce sont des logements plus "commerciaux"
   (gestion professionnelle) vs les hôtes occasionnels.

# PROJET 7 — NosCités | Partie 2 : Requêtes Complexes
# Script Python — PyMongo + Polars
# Association NosCités | Dataset: listings_Paris.csv

# Installation des dépendances :
#   pip install polars pymongo
# ============================================================

import polars as pl
from pymongo import MongoClient

# ============================================================
# OPTION A : Depuis le CSV directement (sans MongoDB)
# ============================================================

print("=" * 60)
print("ANALYSE DIRECTE DEPUIS LE CSV (Polars)")
print("=" * 60)

# Charger le CSV
df = pl.read_csv("listings_Paris.csv", infer_schema_length=10000)
print(f"✅ Dataset chargé : {len(df)} documents, {len(df.columns)} colonnes")

# ── B1. Taux de réservation moyen par type de logement ──────
print("\n── B1. Taux de réservation moyen par mois par type ──")
taux_par_type = (
    df
    .filter(pl.col("reviews_per_month").is_not_null())
    .group_by("room_type")
    .agg([
        pl.col("reviews_per_month").mean().alias("taux_moyen"),
        pl.len().alias("nb_logements")
    ])
    .sort("taux_moyen", descending=True)
)
print(taux_par_type)

# ── B2. Médiane des avis pour tous les logements ─────────────
print("\n── B2. Médiane des avis pour tous les logements ──")
mediane_globale = df["number_of_reviews"].median()
print(f"Médiane globale : {mediane_globale}")
print(f"Moyenne (pour comparaison) : {df['number_of_reviews'].mean():.1f}")
print(f"Max : {df['number_of_reviews'].max()}")
print(f"Sans avis (= 0) : {(df['number_of_reviews'] == 0).sum()} ({100*(df['number_of_reviews'] == 0).sum()/len(df):.1f}%)")

# ── B3. Médiane des avis par catégorie d'hôte ────────────────
print("\n── B3. Médiane des avis par catégorie d'hôte ──")
mediane_par_hote = (
    df
    .group_by("host_is_superhost")
    .agg([
        pl.col("number_of_reviews").median().alias("mediane_avis"),
        pl.col("number_of_reviews").mean().alias("moyenne_avis"),
        pl.len().alias("nb_logements")
    ])
    .with_columns(
        pl.when(pl.col("host_is_superhost") == "t")
          .then(pl.lit("Super hôte"))
          .otherwise(pl.lit("Hôte standard"))
          .alias("categorie")
    )
    .sort("mediane_avis", descending=True)
)
print(mediane_par_hote.select(["categorie", "mediane_avis", "moyenne_avis", "nb_logements"]))

# ── B4. Densité de logements par quartier ────────────────────
print("\n── B4. Top 10 quartiers par densité ──")
densite = (
    df
    .group_by("neighbourhood_cleansed")
    .agg(pl.len().alias("nb_logements"))
    .with_columns(
        (pl.col("nb_logements") / len(df) * 100).round(2).alias("pct_total")
    )
    .sort("nb_logements", descending=True)
    .head(10)
)
print(densite)

# ── B5. Quartiers avec le plus fort taux de réservation ──────
print("\n── B5. Top 10 quartiers par taux de réservation/mois ──")
taux_quartier = (
    df
    .filter(pl.col("reviews_per_month").is_not_null())
    .group_by("neighbourhood_cleansed")
    .agg([
        pl.col("reviews_per_month").mean().alias("taux_moyen"),
        pl.len().alias("nb_annonces")
    ])
    .filter(pl.col("nb_annonces") >= 50)  # filtre statistique
    .sort("taux_moyen", descending=True)
    .head(10)
)
print(taux_quartier)

# ── Bonus : Analyse complète ──────────────────────────────────
print("\n── BONUS : Statistiques globales ──")
print(f"Total annonces      : {len(df)}")
print(f"Hôtes uniques       : {df['host_id'].n_unique()}")
print(f"Super hôtes uniq.   : {df.filter(pl.col('host_is_superhost') == 't')['host_id'].n_unique()}")
print(f"Réserv. instant.    : {(df['instant_bookable'] == 't').sum()} ({100*(df['instant_bookable'] == 't').sum()/len(df):.1f}%)")
print(f"Avec disponibilité  : {(df['has_availability'] == 't').sum()}")

# ============================================================
# OPTION B : Depuis MongoDB avec PyMongo
# ============================================================

print("\n" + "=" * 60)
print("CONNEXION MONGODB (PyMongo)")
print("=" * 60)

try:
    client = MongoClient("mongodb://localhost:27017/", serverSelectionTimeoutMS=3000)
    client.server_info()  # Test de connexion
    db = client["noscites"]
    collection = db["listings"]

    print(f"✅ Connecté à MongoDB | Collection : {collection.count_documents({})} documents")

    # Exemple de requête complexe via PyMongo
    pipeline = [
        {"$match": {"reviews_per_month": {"$type": "double"}}},
        {"$group": {
            "_id": "$neighbourhood_cleansed",
            "taux_moyen": {"$avg": "$reviews_per_month"},
            "nb": {"$sum": 1}
        }},
        {"$match": {"nb": {"$gte": 50}}},
        {"$sort": {"taux_moyen": -1}},
        {"$limit": 5}
    ]

    results = list(collection.aggregate(pipeline))
    print("\nTop 5 quartiers (via MongoDB/PyMongo) :")
    for r in results:
        print(f"  {r['_id']}: {r['taux_moyen']:.3f} avis/mois ({r['nb']} annonces)")

    client.close()

except Exception as e:
    print(f"⚠️  MongoDB non disponible (serveur local) : {e}")
    print("   Utiliser l'Option A (CSV direct) pour les analyses.")

print("\n✅ Script terminé avec succès.")
print("Projet 7 — Association NosCités | Data Engineer | 2024")
