Requête 1
10
Nombre total d'appartements vendus au S1 2020
SELECT COUNT(b.id_mutation) AS nombre_total_appartements_vendus
 FROM bien b
 JOIN mutation m ON b.id_mutation = m.id_mutation
 JOIN type_local tl ON b.code_type_local = tl.code_type_local
 WHERE tl.libelle_type_local = 'Appartement'
 AND m.date_mutation BETWEEN '2020-01-01' AND '2020-06-30';

31 155
appartements vendus
💡 ANALYSE
• Le marché de l'appartement représente le cœur du volume des transactions.
• Le S1 2020 reste dynamique malgré le contexte du début d’année.

Requête 7
16
Évolution des ventes entre T1 et T2 2020
WITH ventes_q1 AS (SELECT COUNT(*) AS nb_ventes_q1 FROM mutation WHERE date_mutation BETWEEN '2020-01-01' AND '2020-03-31'),
 ventes_q2 AS (SELECT COUNT(*) AS nb_ventes_q2 FROM mutation WHERE date_mutation BETWEEN '2020-04-01' AND '2020-06-30')
 SELECT q1.nb_ventes_q1, q2.nb_ventes_q2, q2.nb_ventes_q2 - q1.nb_ventes_q1 AS difference_absolue,
 ROUND(((q2.nb_ventes_q2 - q1.nb_ventes_q1) * 100.0 / q1.nb_ventes_q1), 2) AS taux_evolution_pourcentage
 FROM ventes_q1 q1, ventes_q2 q2;
📊 RÉSULTATS
T1 2020
16 639
ventes
T2 2020
17 279
ventes
+3,85%
+640 ventes
💡 ANALYSE
Contrairement aux attentes,
légère hausse
 du nombre de ventes entre T1 et T2.
Le confinement (17 mars - 11 mai) a décalé les transactions mais le rattrapage au T2 a compensé. Effet de rebond post-confinement.

Requête 8
17
Classement régions - Prix/m² appartements > 4 pièces
SELECT r.nom_region, ROUND(AVG(m.valeur_fonciere / b.surface_reelle_bati), 2) AS prix_m2_moyen, COUNT(*) AS nombre_ventes
 FROM bien b JOIN mutation m ... JOIN type_local tl ... JOIN commune c ... JOIN departement d ... JOIN region r
 WHERE tl.libelle_type_local = 'Appartement' AND b.nombre_pieces_principales > 4

Requête 9
18
Communes avec 50+ ventes au T1 2020
SELECT c.nom_commune AS commune, d.nom_departement AS departement, COUNT(*) AS nombre_ventes_q1
 FROM mutation m JOIN commune c ON m.code_commune = c.code_commune
 JOIN departement d ON c.code_departement = d.code_departement
 WHERE m.date_mutation BETWEEN '2020-01-01' AND '2020-03-31'
 GROUP BY c.code_commune HAVING COUNT(*) >= 50 ORDER BY c.nom_commune;

• Nice : 173 ventes
• Bordeaux : 157 ventes

 de Paris dans le Top 47.

Requête 10
19
Différence de prix/m² entre 2 et 3 pièces
WITH prix_2pieces AS (SELECT AVG(m.valeur_fonciere / b.surface_reelle_bati) AS prix_m2_2p FROM bien b JOIN mutation m ...),
 prix_3pieces AS (SELECT AVG(m.valeur_fonciere / b.surface_reelle_bati) AS prix_m2_3p FROM bien b JOIN mutation m ...)
 SELECT ROUND(p2.prix_m2_2p, 2) AS prix_moyen_2p, ROUND(p3.prix_m2_3p, 2) AS prix_moyen_3p,
 ROUND(p2.prix_m2_2p - p3.prix_m2_3p, 2) AS difference,
 ROUND((p2.prix_m2_2p - p3.prix_m2_3p) * 100.0 / p3.prix_m2_3p, 2) AS difference_pourcentage FROM prix_2pieces p2, prix_3pieces p3;

WITH classement_communes AS (SELECT c.nom_commune, d.code_departement, d.nom_departement,

Top 20 communes - Transactions pour 1000 habitants
SELECT c.nom_commune AS commune, c.population_totale AS population, COUNT(*) AS nombre_transactions,
 ROUND((COUNT(*) * 1000.0) / c.population_totale, 2) AS transactions_pour_1000_habitants
